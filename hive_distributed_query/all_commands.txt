=> CREATE TABLE customer (      // create customer table in /bankdb/customer folder
    customer_id BIGINT,
    name VARCHAR(100),
    email VARCHAR(100)
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/bankdb/customer';

=> desc customer;    // short description about customer table
=> describe formatted customer     // large description about table


## create table with partitioned column
=> CREATE TABLE account (
    account_id BIGINT,
    customer_id BIGINT,
    balance DECIMAL(12,2)
)
PARTITIONED BY (
    branch_id INT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/bankdb/account';

)
